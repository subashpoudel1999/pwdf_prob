import geopandas as gpd
import rasterio

def calibration():
    shapefile_path = '0_shapefiles/1_czma_shapefile/czma_state.shp'

    try:
        # Check State Shapefile
        state_gdf = gpd.read_file(shapefile_path)
        print("\n✅ State Shapefile successfully read")
        print(f"Number of States: {len(state_gdf)}")
        print(f"Shapefile CRS: {state_gdf.crs}")
        print("\nState Names:")
        print(state_gdf['NAME'].tolist())

    except Exception as e:
        print(f"❌ Error in calibration: {e}")

if __name__ == "__main__":
    calibration()
